await page.route("**/api/**", async (route) => {
  const url = new URL(route.request().url());
  const pathname = url.pathname;
  task24.calls.push(pathname + url.search);
  let body;
  let status = 200;
  if (pathname === "/api/staff-session") {
    body = task24.loggedIn ? { success: true, user: task24.db.Staff_Directory[0], permissions: { isAdmin: true, canReadAllAcademicData: true, canWriteAllMarks: true, canRefreshDatabase: true } } : { success: false };
    if (!task24.loggedIn) status = 401;
  } else if (pathname === "/api/database") body = { success: true, data: task24.db, meta: { counts: {} } };
  else if (pathname === "/api/auth/providers") body = { google: { id: "google", name: "Google", type: "oauth", signinUrl: "http://localhost:3014/api/auth/signin/google", callbackUrl: "http://localhost:3014/api/auth/callback/google" } };
  else if (pathname === "/api/auth/csrf") body = { csrfToken: "synthetic-local-only" };
  else if (pathname === "/api/auth/signin/google") { task24.loggedIn = true; body = { url: "http://localhost:3014/?syntheticLogin=1" }; }
  else if (pathname === "/api/auth/signout") { task24.loggedIn = false; body = { url: "http://localhost:3014/" }; }
  else if (pathname === "/api/auth/session") body = null;
  else if (pathname === "/api/marks") {
    const payload = route.request().postDataJSON();
    task24.lastSave = payload;
    if (task24.saveFailure) { status = 400; body = { success: false, error: "Synthetic save failure", requestId: "local-test" }; }
    else {
      for (const record of payload.records) {
        const existing = task24.db.Marks_Log.find((row) => row.Kit_No === record.Kit_No && row.Exam_ID === payload.examId && row.Subject === payload.subject);
        if (existing) existing.Marks_Obtained = record.attendance === "absent" ? "Absent" : record.Marks_Obtained;
      }
      body = { success: true, count: payload.records.length, message: "Synthetic save successful" };
    }
  } else if (pathname === "/api/result-publications") { task24.lastPublication = route.request().postDataJSON(); body = { success: true }; }
  else { status = 404; body = { error: "Unmocked local test API blocked" }; }
  await route.fulfill({ status, contentType: "application/json", body: JSON.stringify(body) });
});

    if (this.download && blobs.has(this.href)) window.__task24Downloads.push({ name: this.download, blob: blobs.get(this.href) });
    else click.call(this);
  };
});
